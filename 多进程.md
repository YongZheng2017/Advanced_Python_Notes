## 多进程

耗CPU的操作：多进程编程
耗IO的操作：多线程编程

&nbsp;

## 模拟CPU操作

```
from concurrent.futures import ThreadPoolExecutor, as_completed
from concurrent.futures import ProcessPoolExecutor
import time

def fib(n):
    if n <= 2:
        return 1
    else:
        return fib(n-1) + fib(n-2)


if __name__ == '__main__':
    with ThreadPoolExecutor(max_workers=3) as executor:
        all_tasks = [executor.submit(fib, (num)) for num in range(25,40)]
        start_time = time.time()
        for future in as_completed(all_tasks):
            data = future.result()
            print("exe result: {}\n".format(data))
        print("Total time: {}".format(time.time()-start_time))

    # Windows系统，一定要放到 if __name__ == '__main__' 下
    with ProcessPoolExecutor(max_workers=3) as executor:
        all_tasks = [executor.submit(fib, (num)) for num in range(25,40)]
        start_time = time.time()
        for future in as_completed(all_tasks):
            data = future.result()
            print("exe result: {}\n".format(data))
        print("Total time: {}".format(time.time()-start_time))
```

## 模拟耗IO操作

```
from concurrent.futures import ThreadPoolExecutor, as_completed
from concurrent.futures import ProcessPoolExecutor
import time

# 用sleep模拟IO操作
def  random_sleep(n):
    time.sleep(n)
    return n


if __name__ == '__main__':
    with ThreadPoolExecutor(max_workers=3) as executor:
        all_tasks = [executor.submit(random_sleep, (num)) for num in [2]*30]
        start_time = time.time()
        for future in as_completed(all_tasks):
            data = future.result()
            print("exe result: {}\n".format(data))
        print("Total time: {}".format(time.time()-start_time))

    # Windows系统，一定要放到 if __name__ == '__main__' 下
    with ProcessPoolExecutor(max_workers=3) as executor:
        all_tasks = [executor.submit(random_sleep, (num)) for num in [2]*30]
        start_time = time.time()
        for future in as_completed(all_tasks):
            data = future.result()
            print("exe result: {}\n".format(data))
        print("Total time: {}".format(time.time()-start_time))
```

## multiprocessing

```
import multiprocessing
import time

def  get_html(n):
    time.sleep(n)
    print("Sub process run finished.")
    return n

if __name__ == '__main__':
    process = multiprocessing.Process(target = get_html, args = (2,))
    process.start()
    print(process.pid)
    process.join()
    print("Main process run finished.")
```
